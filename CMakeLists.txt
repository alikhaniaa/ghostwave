cmake_minimum_required(VERSION 3.20)
project(Ghostwave VERSION 0.2.0 LANGUAGES C CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(GHOSTWAVE_BUILD_GAME "Build the graphical game" ON)
add_library(ghostwave_core src/simulation.cpp src/electrosense.cpp)
target_include_directories(ghostwave_core PUBLIC src)
if(MSVC)
  target_compile_options(ghostwave_core PRIVATE /W4)
else()
  target_compile_options(ghostwave_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(GHOSTWAVE_BUILD_FIELD_LAB "Build the headless electrical experiment exporter" OFF)
if(GHOSTWAVE_BUILD_FIELD_LAB)
  add_executable(field_lab src/field_lab.cpp)
  target_link_libraries(field_lab PRIVATE ghostwave_core)
  if(MINGW)
    target_link_options(field_lab PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  install(TARGETS field_lab RUNTIME DESTINATION .)
  if(MSVC)
    target_compile_options(field_lab PRIVATE /W4)
  else()
    target_compile_options(field_lab PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(ghostwave_tests tests/core_tests.cpp)
  target_link_libraries(ghostwave_tests PRIVATE ghostwave_core)
  add_test(NAME core COMMAND ghostwave_tests)
  add_executable(electrosense_tests tests/electrosense_tests.cpp)
  target_link_libraries(electrosense_tests PRIVATE ghostwave_core)
  add_test(NAME electrosense COMMAND electrosense_tests)
  add_executable(audio_tests tests/audio_tests.cpp)
  target_include_directories(audio_tests PRIVATE tests/audio_support)
  target_link_libraries(audio_tests PRIVATE ghostwave_core)
  if(MSVC)
    target_compile_options(audio_tests PRIVATE /W4)
  else()
    target_compile_options(audio_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME audio COMMAND audio_tests)
  if(TARGET field_lab)
    add_test(NAME field_lab COMMAND field_lab --output-dir "${CMAKE_CURRENT_BINARY_DIR}/field-lab-ctest" --seed 5489)
  endif()
endif()
if(GHOSTWAVE_BUILD_GAME)
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib
    URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz
    URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e)
  FetchContent_MakeAvailable(raylib)
  add_executable(ghostwave src/main.cpp)
  target_link_libraries(ghostwave PRIVATE ghostwave_core raylib)
  if(MINGW)
    target_link_options(ghostwave PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_custom_command(TARGET ghostwave POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:ghostwave>/assets
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/config $<TARGET_FILE_DIR:ghostwave>/config)
  install(TARGETS ghostwave RUNTIME DESTINATION .)
  install(DIRECTORY assets config docs DESTINATION .)
  install(FILES README.md DESTINATION .)
endif()
